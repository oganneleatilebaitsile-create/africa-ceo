export default async function handler(req, res) { /* AFRICA CEO AI BUSINESS ADVISOR Vercel Serverless Function */ // Only allow POST requests if (req.method !== "POST") { return res.status(405).json({ error: "Method not allowed." }); } try { // Get information sent from the website const { message, business } = req.body || {}; // Make sure a question exists if (!message) { return res.status(400).json({ error: "Please provide a business question." }); } // Read the secret API key from Vercel const apiKey = process.env.OPENAI_API_KEY; // Stop if the API key is missing if (!apiKey) { return res.status(500).json({ error: "OPENAI_API_KEY is not configured in Vercel." }); } /* AFRICA CEO SYSTEM INSTRUCTIONS */ const instructions = ` You are AFRICA CEO. You are an AI executive business advisor designed to help African small businesses start, operate, market, grow and scale. Your job is to think like a practical business CEO, strategist, marketing manager, sales manager and operations advisor. IMPORTANT RULES: 1. Give practical answers. 2. Focus on actions the business owner can actually take. 3. Consider African market realities. 4. When appropriate, consider Botswana and Southern African business conditions. 5. Avoid unnecessary corporate jargon. 6. Explain complicated ideas simply. 7. Prioritize low-cost and practical strategies for small businesses. 8. When discussing marketing, give specific content ideas. 9. When discussing sales, give specific actions. 10. When discussing business growth, provide measurable targets. 11. If the user asks for a plan, organize it into clear steps. 12. If useful, provide: - Today - This week - This month - Long-term 13. Do not pretend you performed actions that you cannot actually perform. 14. Be honest when information is missing. 15. The goal is to help the business owner make better decisions and execute faster. BUSINESS INFORMATION: ${JSON.stringify( business || {}, null, 2 )} `; /* Send request to OpenAI Responses API */ const response = await fetch( "https://api.openai.com/v1/responses", { method: "POST", headers: { "Content-Type": "application/json", "Authorization": `Bearer ${apiKey}` }, body: JSON.stringify({ model: "gpt-5.6-luna", instructions: instructions, input: message, max_output_tokens: 1000 }) } ); /* Convert OpenAI response into JSON */ const data = await response.json(); /* Handle OpenAI errors */ if (!response.ok) { console.error( "OpenAI error:", data ); return res.status( response.status ).json({ error: data?.error?.message || "OpenAI request failed." }); } /* Return AI answer */ const answer = data.output_text || extractOutputText(data); return res.status(200).json({ answer: answer || "No response was returned by the AI CEO." }); } catch (error) { console.error( "Africa CEO server error:", error ); return res.status(500).json({ error: error.message || "Africa CEO server error." }); } } /* Backup extractor in case output_text is not available. */ function extractOutputText(data) { try { const output = data?.output || []; let text = ""; for ( const item of output ) { if ( item.type === "message" && Array.isArray(item.content) ) { for ( const content of item.content ) { if ( content.type === "output_text" ) { text += content.text || ""; } } } } return text.trim(); } catch (error) { return ""; } }
